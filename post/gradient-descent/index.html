代码示例

设函数为f(x)=3x2+4x+5f(x) = 3x^{2} + 4x + 5f(x)=3x2+4x+5， 则ddxf(x)=6x+4\frac{\mathrm d}{\mathrm dx} f(x) = 6x + 4dxd​f(x)=6x+4， 最低点为x_min=−B2A=−23x\_{min} = - \frac{B}{2A} = - \frac{2}{3}x_min=−2AB​=−32​。

#!/usr/bin/env python3

import numpy as np

a = 3
b = 4
c = 5

# f(x) = ax^2 + bx + c
def f(x):
    return a*(x**2) + b*x + c

# df(x) = 2ax + b
def df(x):
    return 2*a*x + b

def main():
    # 最低点
    xmin = - b/(2*a)
    # 步长
    alpha = 0.1
    # 随机初始值
    x = np.random.randint(5)
    # 递减 100 次
    for i in range(100):
        x = x - alpha * df(x)

    # 输出结果
    print("x_n is:" , x)
    print("x_min is:", xmin)

if __name__ == "__main__":
    main()
